cmake_minimum_required(VERSION 3.20)
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
  set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/pico_sdk_init.cmake")
  message(FATAL_ERROR "Set PICO_SDK_PATH to the pinned SDK in dependencies.json")
endif()
set(PICO_BOARD ducktop2_maker)
set(PICO_BOARD_HEADER_DIRS ${CMAKE_CURRENT_LIST_DIR})
set(PICO_PLATFORM rp2350-arm-s)
set(PICO_NO_PICOTOOL 1)
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(ducktop2_maker C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(ducktop2_maker main.c bus_target.c usb_descriptors.c ../maker/src/maker_policy.c ../maker/src/maker_bus.c)
target_include_directories(ducktop2_maker PRIVATE . ../maker/include)
target_link_libraries(ducktop2_maker pico_stdlib hardware_watchdog hardware_adc hardware_uart hardware_i2c hardware_spi tinyusb_device tinyusb_board)
target_compile_options(ducktop2_maker PRIVATE -Wall -Wextra -Werror -fstack-usage)
pico_enable_stdio_uart(ducktop2_maker 0)
pico_enable_stdio_usb(ducktop2_maker 0)
pico_add_extra_outputs(ducktop2_maker)
